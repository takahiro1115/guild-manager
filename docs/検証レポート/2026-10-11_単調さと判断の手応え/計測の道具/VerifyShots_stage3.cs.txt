// 検証用（サンドボックスだけ）：段3（§0.99）の「まとめて」の画面を撮る。
using Godot;
using System;
using System.Collections.Generic;
using System.Linq;
using System.Reflection;
using System.Threading.Tasks;
using GuildManager.Core.Models;
using GuildManager.Core.Rng;
using GuildManager.Core.Systems;

public partial class VerifyShots : Node
{
	private string _out = "";
	private int _n;

	public override async void _Ready()
	{
		_out = OS.GetEnvironment("SHOT_DIR");
		if (string.IsNullOrEmpty(_out)) return;
		DirAccess.MakeDirRecursiveAbsolute(_out);
		await Frames(60);
		try { await Stage3(); }
		catch (Exception e) { GD.PrintErr("VERIFY ERROR: " + e); }
		GD.Print("VERIFY DONE");
		GetTree().Quit();
	}

	private async Task Frames(int n)
	{
		for (int i = 0; i < n; i++) await ToSignal(GetTree(), SceneTree.SignalName.ProcessFrame);
	}

	private async Task Shot(string name)
	{
		await Frames(15);
		var img = GetViewport().GetTexture().GetImage();
		string path = $"{_out}/{++_n:00}_{name}.png";
		img.SavePng(path);
		GD.Print($"SHOT {path}");
	}

	/// <summary>小窓も入れて撮る（小窓は別ウィンドウなので、その窓の絵を撮る）。</summary>
	private async Task ShotWindow(Window w, string name)
	{
		await Frames(15);
		var img = w.GetTexture().GetImage();
		string path = $"{_out}/{++_n:00}_{name}.png";
		img.SavePng(path);
		GD.Print($"SHOT {path}");
	}

	private MainDashboard Dash => GetTree().Root.GetChildren().OfType<MainDashboard>().First();

	private object? Call(string method, params object[] args)
	{
		var mi = typeof(MainDashboard).GetMethod(method, BindingFlags.NonPublic | BindingFlags.Instance | BindingFlags.Public);
		return mi?.Invoke(Dash, args);
	}

	private static object? CallOn(object target, string method, params object?[] args) =>
		target.GetType().GetMethod(method, BindingFlags.NonPublic | BindingFlags.Instance | BindingFlags.Public)?.Invoke(target, args);

	private T Field<T>(object target, string name) =>
		(T)target.GetType().GetField(name, BindingFlags.NonPublic | BindingFlags.Instance)!.GetValue(target)!;

	private async Task SwitchView(string view)
	{
		var enumType = typeof(MainDashboard).GetNestedType("DashboardView", BindingFlags.NonPublic)!;
		Call("SwitchView", Enum.Parse(enumType, view));
		await Frames(10);
	}

	private List<Window> OpenWindows() =>
		AllNodes(GetTree().Root).OfType<Window>().Where(w => w != GetTree().Root && w.Visible).ToList();

	private static IEnumerable<Node> AllNodes(Node n)
	{
		foreach (var c in n.GetChildren())
		{
			yield return c;
			foreach (var d in AllNodes(c)) yield return d;
		}
	}

	private async Task PressAndConfirm(Button button, string name)
	{
		GD.Print($"BUTTON {name}: '{button.Text}' disabled={button.Disabled} visible={button.Visible} tip='{button.TooltipText}'");
		button.EmitSignal(BaseButton.SignalName.Pressed);
		await Frames(20);
		var dialog = OpenWindows().OfType<ConfirmationDialog>().LastOrDefault();
		if (dialog == null) { GD.Print($"NO DIALOG for {name}"); return; }
		await ShotWindow(dialog, name + "_confirm");
		dialog.EmitSignal(AcceptDialog.SignalName.Confirmed);
		await Frames(20);
		await Shot(name + "_done");
	}

	private async Task Stage3()
	{
		await Frames(30);
		OpenWindows().OfType<ConfirmationDialog>().FirstOrDefault()?.EmitSignal(AcceptDialog.SignalName.Confirmed);
		await Frames(30);
		for (int i = 0; i < 5 && OpenWindows().Count > 0; i++)
		{
			var w = OpenWindows().Last();
			if (w is AcceptDialog ad) ad.EmitSignal(AcceptDialog.SignalName.Confirmed); else w.EmitSignal(Window.SignalName.CloseRequested);
			await Frames(20);
		}

		var state = Field<GameState>(Dash, "_state");
		// 撮るための在庫を足す：遺物（銅3・銀2・金1）、保管庫の武具（店の各段の品と、弱い品）、余った素材、お金
		var rng = new SeededRng(11);
		foreach (var r in new[] { ItemRarity.Common, ItemRarity.Common, ItemRarity.Common, ItemRarity.Rare, ItemRarity.Rare, ItemRarity.Epic })
		{
			var relic = AppraisalSystem.CreateRelic(rng, "forest", 30);
			relic.Rarity = r;
			relic.AppraisalCost = GuildManager.Core.Balance.RelicBalance.GetAppraisalCost(r);
			state.UnidentifiedItems.Add(relic);
		}
		foreach (var item in ItemCatalog.GetAll().Where(i => i.ShopTier is 0 or 3))
			state.Armory.Add(EquipmentItem.FromCatalog(item, state.WeekNumber, "検証"));
		state.AddMaterial("mat_forest_herb", 40);
		state.AddMaterial("mat_cave_moss", 12);
		state.Gold += 40000;
		// 新人戦のおすすめを撮るため、控えの2人と潜行の部隊の1人をこの年に加わったことにする
		var diveIds = state.SavedParties.Where(p => p.Order == SquadOrder.Dive).SelectMany(p => p.MemberIds).ToHashSet();
		int year = GameCalendar.YearOf(state.WeekNumber);
		foreach (var a in state.Adventurers.Where(a => !diveIds.Contains(a.Id)).Take(2).Concat(state.Adventurers.Where(a => diveIds.Contains(a.Id)).Take(1)))
			a.JoinedYear = year;
		Call("RefreshAll");
		GD.Print($"STATE week={state.WeekNumber} gold={state.Gold} relics={state.UnidentifiedItems.Count} armory={state.Armory.Count}");

		// 倉庫：まとめて鑑定・まとめて売る
		await SwitchView("Warehouse");
		await Shot("warehouse");
		var inv = Field<InventoryPanel>(Dash, "_inventoryPanel");
		await PressAndConfirm(Field<Button>(inv, "_bulkAppraiseButton"), "warehouse_bulk_appraise");
		await PressAndConfirm(Field<Button>(inv, "_bulkSellButton"), "warehouse_bulk_sell");

		// 部隊：おすすめを着せる（潜行の部隊を選ぶ）
		await SwitchView("Party");
		var party = Field<PartyFormationPanel>(Dash, "_partyFormationPanel");
		int idx = state.SavedParties.Take(4).ToList().FindIndex(p => p.Order == SquadOrder.Dive);
		CallOn(party, "OnSquadSelected", Math.Max(0, idx));
		await Frames(10);
		await Shot("party");
		await PressAndConfirm(Field<Button>(party, "_recommendGearButton"), "party_recommend_gear");

		// 大会：おすすめの子で申し込む
		await SwitchView("Tournament");
		await Shot("tournament");
		var tp = Field<TournamentPanel>(Dash, "_tournamentPanel");
		var rec = Field<Button>(tp, "_recommendButton");
		if (rec.Visible) await PressAndConfirm(rec, "tournament_recommend");
		else GD.Print("TOURNAMENT no recommendation this month");

		// システム：交渉の自動
		await SwitchView("System");
		var sys = Field<SystemPanel>(Dash, "_systemPanel");
		Field<CheckButton>(sys, "_autoBonusCheck").ButtonPressed = true;
		await Shot("system_auto_bonus");
		GD.Print($"AUTO BONUS = {state.AutoNegotiationBonus}");

		// 月報：今月の遺物
		await SwitchView("Dungeon");
		var popup = Field<MonthlyReportPopup>(Dash, "_monthlyReportPopup");
		for (int m = 1; m <= 3; m++)
		{
			Call("OnNextMonthPressed");
			for (int i = 0; i < 12; i++)
			{
				await Frames(20);
				var wins = OpenWindows();
				if (wins.Count == 0) break;
				var w = wins.Last();
				if (w == popup)
				{
					var report = state.MonthlyReports.LastOrDefault();
					GD.Print($"MONTH {m} relicIds={report?.RelicIds.Count} highlights={string.Join(" / ", report?.Highlights.Select(h => h.Text) ?? Array.Empty<string>())}");
					await ShotWindow(popup, $"m{m}_report");
					var bulk = AllNodes(popup).OfType<Button>().FirstOrDefault(b => b.Text.Contains("まとめて鑑定") && !b.Disabled);
					var single = AllNodes(popup).OfType<Button>().FirstOrDefault(b => b.Text.StartsWith("鑑定する") && !b.Disabled);
					if (bulk != null || single != null)
					{
						(bulk ?? single)!.EmitSignal(BaseButton.SignalName.Pressed);
						await Frames(20);
						await ShotWindow(popup, $"m{m}_report_appraised");
					}
				}
				else await Shot($"m{m}_{w.GetType().Name}");
				if (w is ConfirmationDialog cd) cd.EmitSignal(AcceptDialog.SignalName.Confirmed);
				else if (w is AcceptDialog ad) ad.EmitSignal(AcceptDialog.SignalName.Confirmed);
				else w.EmitSignal(Window.SignalName.CloseRequested);
			}
		}
	}
}
